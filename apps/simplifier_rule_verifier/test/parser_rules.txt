# These rules are only true if the parser gets precedence and associativity
# right, so they pin down the expression grammar.

# Multiplicative binds tighter than additive
rewrite(1 + 2*3, 7)
rewrite((1 + 2)*3, 9)

# Additive and multiplicative are left-associative
rewrite(1 - 2 - 3, -4)
rewrite(12 / 3 / 2, 2)
rewrite(2 * 3 % 4, 2)

# Comparisons bind looser than arithmetic
rewrite(1 + 2 < 4, true)
rewrite(2 * 3 == 6, true)

# && binds tighter than ||
rewrite(x < y && y < z || x < z, x < z)

# Unary operators bind tighter than any binary one
rewrite(-x + y, y - x)
rewrite(-x * 2, x * -2)
rewrite(!(x < y), y <= x)

# Integers of a type other than Int(32) are printed with a suffix giving the
# type. These rules are only true if the suffix is parsed; they are false if
# the literals are taken to be Int(32).
rewrite(255_u8 + 1_u8, 0_u8)
rewrite(65535_u16 + 1_u16, 0_u16)
rewrite(32767_i16 + 1_i16, -32768_i16)
rewrite(-3_i16 * 2_i16, -6_i16)

# Bitwise not
rewrite(~0_i16 + 1_i16, 0_i16)
